cmake_minimum_required(VERSION 3.16)
project(tase_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -fno-fast-math)
add_library(tase_core tase_core.cpp)
target_include_directories(tase_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(tase_core_replay replay_main.cpp)
target_link_libraries(tase_core_replay PRIVATE tase_core)
add_executable(tase_core_test test_tase_core.cpp)
target_link_libraries(tase_core_test PRIVATE tase_core)
enable_testing()
add_test(NAME tase_core COMMAND tase_core_test)
add_test(NAME tase_core_replay COMMAND tase_core_replay)
